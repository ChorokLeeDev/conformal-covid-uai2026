\begin{frame}[t]{Diagnosing Conformal Prediction Failures}
\begin{itemize}
\item Under Distribution Shift: A COVID-19 Case Study
\item Chorok Lee · KAIST · UAI 2026

\end{itemize}
\deckpoint{What does the model rely on?}
\end{frame}
% Slide 2: Setup —
\begin{frame}[t]{Setup: feature row to label set}
\deckstage{Setup}{}
\begin{itemize}
\item Input features $\rightarrow$ LightGBM $\rightarrow$ class probabilities
\item APS uses probabilities and a labelled calibration sample to make a label set.
\item Coverage: how often the set includes the true label.
\end{itemize}
\deckpoint{Feature row, class probabilities, label set.}
\end{frame}
% Slide 3: Assumptions —
\begin{frame}[t]{Assumptions: what must stay stable?}
\deckstage{Assumptions}{}
\begin{itemize}
\item The conformal promise needs exchangeable calibration and future cases, with independent model fitting.
\item The tasks share COVID periods, not identical distributions.
\item The mechanism theorem assumes a damaged probability component, a stable remaining component and a fixed threshold.
\end{itemize}
\deckpoint{The promise and the mechanism each have assumptions.}
\end{frame}
% Slide 4: Hypothesis —
\begin{frame}[t]{Hypothesis: can reliance help explain loss?}
\deckstage{Hypothesis}{}
\begin{itemize}
\item SHAP explains the fitted model’s output through feature contributions.
\item Concentration: the top feature’s average absolute contribution divided by the total.
\item Could a change in that feature’s predictive relationship help explain coverage loss?
\end{itemize}
\deckpoint{The changed relationship matters.}
\end{frame}

% Slide 7: Theorem —
\begin{frame}[t]{Theorem: a conditional explanation}
\deckstage{Theorem}{}
\begin{itemize}
\item A simplified model combines two probability components.
\item One becomes unreliable after shift; the other stays stable.
\item Under the stated conditions, weighting the damaged component more worsens bounds on the APS score and coverage.
\end{itemize}
\deckpoint{A bound for a simplified model.}
\end{frame}
% Slide 8:  —
\begin{frame}[t]{Limits: what is still unvalidated?}
\begin{itemize}
\item The historical 40\% cutoff is not a validated deployment rule.
\item The positive historical association was not reproduced in the separate later panel.
\item Measured SHAP concentration and a theoretical mixture share are different objects.
\end{itemize}
\deckpoint{What evidence would support a stronger claim?}
\end{frame}
% Slide 9:  —
\begin{frame}[t]{Takeaway: ask what the evidence supports}
\begin{itemize}
\item Start with the classifier, its output and its prediction set.
\item Explain the assumptions before interpreting a failure.
\item Treat attribution as a measured clue with a stated scope.
\end{itemize}
\deckpoint{What does the evidence establish?}
\end{frame}
