% ====================== 2. MOTIVATION ======================
\begin{frame}{Why this matters: a guarantee that quietly breaks}
  \begin{itemize}
    \item \key{Conformal prediction} turns any model into one that outputs a
          \emph{prediction set} with a coverage guarantee --- e.g.\ ``the true
          label is inside the set \obs{90\%} of the time.''
    \item That guarantee holds only under \key{exchangeability}: test data looks
          like calibration data. Real deployments face \key{distribution shift}.
    \vspace{4pt}
    \item Prior work characterizes \emph{how much} coverage degrades under shift.
          The open gap:
  \end{itemize}
  \vspace{2pt}
  \begin{center}
  \setbeamercolor{block body}{bg=lblue!40}
  \begin{beamercolorbox}[wd=0.92\textwidth,sep=8pt,rounded=true]{block body}
    \centering\bfseries
    Can we identify \emph{which} deployed models will fail\\
    \est{before observing any test data?}
  \end{beamercolorbox}
  \end{center}
  \vspace{2pt}
  \begin{itemize}
    \item Our case study: \hl{8 tasks}, one \emph{identical} COVID-19 shift, yet
          coverage drops span \obs{0.1\,pp} to \hl{77.1\,pp}. Same shift, opposite outcomes.
  \end{itemize}
\end{frame}

% ====================== 3. PROBLEM SETUP ======================
\begin{frame}{Problem setup: the COVID-19 natural experiment}
  \slab{A.\ Data.} SALT supply-chain benchmark (RelBench), with a temporal split
  aligned to COVID-19:
  \begin{itemize}
    \item \obs{Train}: before Feb 2020 (pre-COVID)\quad
          \obs{Validation}: Feb--Jul 2020 (onset)\quad
          \obs{Test}: after Jul 2020 (peak)
    \item \key{SHAP $C$ and the initial 40\% cutoff} use \obs{validation-time data only}.
          Test observations support held-out evaluation and post-hoc
          mechanism/threshold analyses; they do not tune that initial cutoff.
  \end{itemize}
  \vspace{4pt}
  \slab{B.\ Model + conformal method.} LightGBM classifier $+$ Adaptive Prediction
  Sets (APS), target coverage $1-\alpha = \obs{90\%}$.
  \vspace{4pt}
  \slab{C.\ What ``failure'' means.} \key{Coverage drop} $=$ validation coverage
  $-$ test coverage (percentage points). A big drop $=$ the guarantee broke.
  \vspace{6pt}
  \begin{center}\footnotesize
    All 8 tasks pass through the \emph{same} pipeline and the \emph{same} shift ---
    the only thing that varies is the task.
  \end{center}
\end{frame}

% ====================== 4. METHOD ======================
\begin{frame}{Method: SHAP concentration as a pre-deployment diagnostic}
  \key{Idea.} A model that leans on \emph{one} feature is fragile: if that feature
  shifts, everything shifts. We measure that reliance \est{before deployment},
  on \obs{validation data only} --- no test labels.
  \vspace{6pt}

  % ---- Diagnostic flow as plain text (no drawn schematic) ----
  \begin{center}
  \setbeamercolor{block body}{bg=lblue!40}
  \begin{beamercolorbox}[wd=0.95\textwidth,sep=8pt,rounded=true]{block body}
    \centering\small
    \obs{\textbf{Validation data}} {\color{pgrey}(no test labels)}
    \;$\longrightarrow$\;
    \key{SHAP} importance $\phi_j=\text{mean}|\text{SHAP}|$
    \;$\longrightarrow$\;
    \key{Concentration} $C=\dfrac{\phi_1}{\sum_{j}\phi_j}$
    \;$\longrightarrow$\;
    \est{\textbf{40\% diagnostic}}
  \end{beamercolorbox}
  \end{center}
  \vspace{4pt}
  \begin{itemize}\small
    \item $C \in [0,1]$: fraction of total importance in the single top feature
          $\phi_1$. Normalized, so it \emph{compares across tasks}.
    \item \key{Decision rule (exploratory):}\; $C \le \obs{40\%}$ lower risk\;\textbullet\;
          $C > \obs{40\%}$ check protective factors, else flag \est{vulnerable}.
  \end{itemize}
\end{frame}

% ====================== 5. THEORY ======================
\begin{frame}{Theory: why concentration relates to failure}
  \key{Theorem 1 (monotone vulnerability).} For a $K$-class APS predictor,
  under three explicit assumptions ---
  \begin{itemize}\small
    \item[(A1)] \emph{additive decomposition}: prediction $=C\cdot g(y\mid x_1)+(1-C)\,h(\cdot)$
    \item[(A2)] \emph{concentrated misclassification} under shift on the top feature $x_1$
    \item[(A3)] \emph{residual exchangeability} of the remaining features
  \end{itemize}
  the \emph{lower bound} on the expected conformity score increases with $C$, and
  the coverage \emph{upper bound} is \key{non-increasing} in $C$, on the stated
  domain ($\varepsilon<\bar h$ and $\varepsilon<(1-\hat q_\alpha)/(K-1)$, $K\ge3$):
  \begin{equation*}
    \mathbb{E}[s_{\text{test}}] \;\geq\;
      C\,(1-(K{-}1)\varepsilon) \;+\; (1-C)\,(1-(K{-}1)\bar h)
    \eqnum{2}
  \end{equation*}
  \begin{itemize}
    \item A \est{directional intuition} for an \emph{idealized probability-mixture}
          model --- \est{not} a formal guarantee for the empirical SHAP-derived $C$
          (TreeSHAP log-odds; footnote 3, p.4).
    \item \obs{Empirically} consistent: catastrophic tasks show strong rightward
          score shifts (KS $0.68$--$0.96$).
  \end{itemize}
\end{frame}

% ====================== 6. EXPERIMENTS / MECHANISM ======================
\begin{frame}{The mechanism: catastrophic vs.\ robust, same shift}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \slab{Catastrophic} (s-shipcond, \obs{71.6\,pp} drop)
      \begin{itemize}\small
        \item Top feature \key{explodes} ($4.5\times$ importance).
        \item Sets \emph{shrink} while they should grow:
              \est{confidently wrong}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \slab{Robust} (s-office, \obs{0.1\,pp} drop, yet $C=42.6\%$)
      \begin{itemize}\small
        \item Importance \key{redistributes} across many
              features ($10$--$20\times$).
        \item Rankings reshuffle gracefully; coverage holds ---
              a \est{high-$C$ false positive}.
      \end{itemize}
    \end{column}
  \end{columns}

    $\rho\le\obs{0.19}$.
  \end{center}

% ====================== 8. LIMITATIONS (honest) ======================
\begin{frame}{Limitations: what this diagnostic does \emph{not} do}
  \slab{A.\ It is model-specific.} The signal is strong for gradient-boosted trees
  (LightGBM $\rho=\obs{0.833}$, CatBoost $0.67$, XGBoost $0.55$) but \est{fails for
  bagging}: Random Forest $\rho=\obs{0.30}$ ($p=0.47$). Compute $C$ on \emph{your}
  deployed model class, don't transfer.
  \vspace{4pt}
  \slab{B.\ The 40\% threshold is exploratory.} Derived from \obs{8} SALT tasks;
  within-SALT power is below convention. Treat as provisional; validate before a
  new domain.
  \vspace{4pt}
  \slab{C.\ One false positive, honestly reported.} sales-office has $C=42.6\%$ yet
  \obs{0.1\,pp} drop. The framework reports protective SALESORGANIZATION
  (train--validation Jaccard $=.61$; treated as secondary), motivating a
  \key{provisional protective-factor check}, not a hard cutoff.
  \vspace{4pt}
  \slab{D.\ It diagnoses, it does not fix.} Mitigation is mechanism-dependent:
  quarterly retraining gives \obs{$+18.9$\,pp} on \emph{one} task (s-shipcond),
  \est{single-seed, Holm-corrected $p=0.11$} --- suggestive, not a reliable average.
  RAPS helps class-accumulation failures, not single-feature dependence.
\end{frame}

% ====================== 9. TAKEAWAY ======================
\begin{frame}{Takeaway}
  \vspace{4pt}
  \begin{enumerate}
    \item \key{The failure is silent.} Under shift, catastrophic tasks become
          \est{confidently wrong} --- prediction sets shrink while coverage
          collapses. Standard shift detectors cannot see this.
    \vspace{6pt}
    \item \key{One number, computed before deployment.} SHAP concentration
          $C=\phi_1/\sum\phi_j$, on validation data only, is associated with
          failure severity: \hl{$\rho=0.853$}, $p<0.001$ across 16 tasks, 9 domains.
    \vspace{6pt}
    \item \key{Honest scope.} Specific to gradient-boosted models with genuine
          single-feature dependence; the threshold is exploratory; it flags risk,
          it does not repair it.
  \end{enumerate}
